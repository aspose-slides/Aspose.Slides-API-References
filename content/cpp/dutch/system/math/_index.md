## Opmerkingen



```cpp
#include "system/math.h"
#include <iostream>

int main()
{
  using namespace System;

  // Print de absolute waarden.
  for (int i = -1; i < 2; ++i)
  {
    std::cout << Math::Abs(i) << " ";
  }
  std::cout << std::endl;

  // Print de sinus van PI/2 en de cosinus van PI.
  std::cout << "sin(PI/2)=" << Math::Sin(Math::PI/2) << "; cos(PI)=" << Math::Cos(Math::PI) << std::endl;

  return 0;
}
/*
Dit codevoorbeeld produceert de volgende uitvoer:
1 0 1
sin(PI/2)=1; cos(PI)=-1
*/
```
