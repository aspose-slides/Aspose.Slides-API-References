### أمثلة

المثال التالي يوضح كيفية الوصول إلى إطارات كائن OLE.

```csharp
[C#]
// يحمّل ملف PPTX إلى كائن عرض تقديمي
using (Presentation pres = new Presentation("AccessingOLEObjectFrame.pptx"))
{
    // الوصول إلى الشريحة الأولى
    ISlide sld = pres.Slides[0];
    // تحويل الشكل إلى OleObjectFrame
    OleObjectFrame oleObjectFrame = sld.Shapes[0] as OleObjectFrame;
    // قراءة كائن OLE وكتابته إلى القرص
    if (oleObjectFrame != null)
    {
        // جلب بيانات الملف المدمج
        byte[] data = oleObjectFrame.EmbeddedData.EmbeddedFileData;
        // جلب امتداد الملف المدمج
        string fileExtention = oleObjectFrame.EmbeddedData.EmbeddedFileExtension;
        // إنشاء مسار لحفظ الملف المستخرج
        string extractedPath = "excelFromOLE_out" + fileExtention;
        // حفظ البيانات المستخرجة
        using (FileStream fstr = new FileStream(extractedPath, FileMode.Create, FileAccess.Write))
        {
            fstr.Write(data, 0, data.Length);
        }
    }
}
```
